# 按条件拿某个字段值
curl "http://127.0.0.1:8858/api/sqlite_demo/foose_users/value/username?id=1" \
  -H "Authorization: Bearer $TOKEN"
# → "demo"

# 用 LIKE 运算符
curl "http://127.0.0.1:8858/api/sqlite_demo/foose_users/value/id?username[_like]=admin%" \
  -H "Authorization: Bearer $TOKEN"
# → 1







TOKEN="<你的 token>"
OBJECT="sqlite_demo"
TABLE="foose_users"
BASE="http://127.0.0.1:8858"

# ① 单字段自增 +1（默认）
curl -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?id=1" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"inc":{"flag":1}}'

# ← {"ok":true,"changed":1,"rows":[{"id":1,"flag":1,...}]}

# ② 单字段自增 +100（自定义增量）
curl -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?id=1" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"inc":{"flag":100}}'

# ③ 单字段自减 -5
curl -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?id=1" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"dec":{"flag":5}}'

# ④ 同时 inc + dec 多个字段（推荐方式）
curl -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?id=1" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "inc": { "flag": 2, "score": 10 },
    "dec": { "stock": 5 }
  }'

#    flag += 2, score += 10, stock -= 5（三条 SET 同事务执行）

# ⑤ 同一个字段同时出现在 inc 和 dec — 合并代数和
curl -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?id=1" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"inc":{"flag":3},"dec":{"flag":1}}'

#    实际等效 flag += 2（3 - 1），零开销

# ⑥ 多行全部更新 — 匹配条件命中几行改几行
curl -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?flag=0" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"inc":{"flag":1}}'

# ← changed=N，rows 返回所有被更新行的最新数据

# ⑦ 配合 Directus 风格 filter 运算符
curl -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?username[_like]=admin%25&flag=0" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"inc":{"flag":1}}'

# ⑧ 延迟执行 delay=1500ms（先 sleep 再 UPDATE，上限 30000）
curl -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?id=1&delay=1500" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"inc":{"flag":1}}'

# ← 响应耗时 ≥ 1.5s

# ⑨ error 示例 — 非数值字段
curl -s -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?id=1" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"inc":{"username":1}}'

# ← {"error":"字段 \"username\" 类型 \"TEXT\" 不是数值型，无法自增/自减"}

# ⑩ error 示例 — 无 filter（防误改整表）
curl -s -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"inc":{"flag":1}}'

# ← {"error":"至少需要一个查询条件来定位要更新的行"}

# ⑪ error 示例 — 空 body
curl -s -X POST "$BASE/api/$OBJECT/$TABLE/inc-dec?id=1" \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{}'

# ← {"error":"inc 和 dec 至少要有一个字段"}
